5)CREATE TABLE public.chat_participants (
    id BIGINT GENERATED BY DEFAULT AS IDENTITY PRIMARY KEY,

    environment VARCHAR(20) NOT NULL,

    match_id INTEGER DEFAULT NULL,

    gaming_match_id INTEGER DEFAULT NULL,

    user_id INTEGER NOT NULL,

    created_at TIMESTAMPTZ NOT NULL DEFAULT NOW(),

    CONSTRAINT chat_participants_match_check
        CHECK (
            (match_id IS NOT NULL AND gaming_match_id IS NULL)
            OR
            (match_id IS NULL AND gaming_match_id IS NOT NULL)
        ),

    CONSTRAINT chat_participants_environment_check
        CHECK (
            environment IN ('local', 'production')
        ),

    CONSTRAINT chat_participants_unique_offline
        UNIQUE (environment, match_id, user_id),

    CONSTRAINT chat_participants_unique_gaming
        UNIQUE (environment, gaming_match_id, user_id)
);

6)CREATE TABLE public.chat_messages (
    id BIGINT GENERATED BY DEFAULT AS IDENTITY PRIMARY KEY,

    environment VARCHAR(20) NOT NULL,

    match_id INTEGER DEFAULT NULL,

    gaming_match_id INTEGER DEFAULT NULL,

    user_id INTEGER NOT NULL,

    message TEXT NOT NULL,

    created_at TIMESTAMPTZ NOT NULL DEFAULT NOW(),

    CONSTRAINT chat_messages_match_check
        CHECK (
            (match_id IS NOT NULL AND gaming_match_id IS NULL)
            OR
            (match_id IS NULL AND gaming_match_id IS NOT NULL)
        ),

    CONSTRAINT chat_messages_message_length_check
        CHECK (
            char_length(btrim(message)) BETWEEN 1 AND 500
        ),

    CONSTRAINT chat_messages_environment_check
        CHECK (
            environment IN ('local', 'production')
        )
);

//authorization function
CREATE FUNCTION public.is_chat_participant(
    p_user_id INTEGER,
    p_environment VARCHAR,
    p_match_id INTEGER,
    p_gaming_match_id INTEGER
)
RETURNS BOOLEAN
LANGUAGE SQL
STABLE
SECURITY DEFINER
SET search_path = public
AS $$
    SELECT EXISTS (
        SELECT 1
        FROM public.chat_participants cp
        WHERE cp.user_id = p_user_id
          AND cp.environment = p_environment
          AND (
              (
                  p_match_id IS NOT NULL
                  AND cp.match_id = p_match_id
              )
              OR
              (
                  p_gaming_match_id IS NOT NULL
                  AND cp.gaming_match_id = p_gaming_match_id
              )
          )
    );
$$;

Then:
REVOKE ALL
ON FUNCTION public.is_chat_participant(
    INTEGER,
    VARCHAR,
    INTEGER,
    INTEGER
)
FROM PUBLIC;

GRANT EXECUTE
ON FUNCTION public.is_chat_participant(
    INTEGER,
    VARCHAR,
    INTEGER,
    INTEGER
)
TO authenticated;



//environment trigger
CREATE FUNCTION public.set_chat_message_environment()
RETURNS TRIGGER
LANGUAGE plpgsql
SECURITY INVOKER
SET search_path = public
AS $$
DECLARE
    jwt_environment TEXT;
BEGIN
    jwt_environment := auth.jwt() ->> 'environment';

    IF jwt_environment NOT IN ('local', 'production') THEN
        RAISE EXCEPTION 'Invalid chat environment';
    END IF;

    NEW.environment := jwt_environment;

    RETURN NEW;
END;
$$;

Then:
CREATE TRIGGER trg_set_chat_message_environment
BEFORE INSERT
ON public.chat_messages
FOR EACH ROW
EXECUTE FUNCTION public.set_chat_message_environment();


//final INSERT policy
CREATE POLICY "Authorized users can send chat messages"
ON public.chat_messages
FOR INSERT
TO authenticated
WITH CHECK (
    chat_messages.user_id = (
        auth.jwt() ->> 'sportilo_user_id'
    )::INTEGER

    AND

    public.is_chat_participant(
        (
            auth.jwt() ->> 'sportilo_user_id'
        )::INTEGER,

        (
            auth.jwt() ->> 'environment'
        )::VARCHAR,

        chat_messages.match_id,
        chat_messages.gaming_match_id
    )
);

//final SELECT policy
CREATE POLICY "Read chat messages for authorized chat"
ON public.chat_messages
FOR SELECT
TO anon, authenticated
USING (
    (SELECT auth.jwt() ->> 'chat_access') = 'read'

    AND

    (SELECT auth.jwt() ->> 'environment') = environment

    AND

    (
        (
            (SELECT auth.jwt() ->> 'chat_type') = 'offline'
            AND
            match_id = (
                (SELECT auth.jwt() ->> 'chat_id')
            )::INTEGER
        )

        OR

        (
            (SELECT auth.jwt() ->> 'chat_type') = 'gaming'
            AND
            gaming_match_id = (
                (SELECT auth.jwt() ->> 'chat_id')
            )::INTEGER
        )
    )
);

Then:
GRANT SELECT
ON public.chat_messages
TO anon, authenticated;


//INSERT permission

GRANT INSERT
ON public.chat_messages
TO authenticated;
