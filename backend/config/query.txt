1) user table query

CREATE TABLE users (
    id SERIAL PRIMARY KEY,

    name VARCHAR(100) NOT NULL,

    email VARCHAR(100) UNIQUE NOT NULL,

    password TEXT,

    provider VARCHAR(20) NOT NULL,

    created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP
);


2)create offline match table query

CREATE TABLE matches (
    id SERIAL PRIMARY KEY,

    -- ADMIN INFO
    admin_name VARCHAR(100) NOT NULL,
    admin_email VARCHAR(100) NOT NULL,

    -- MATCH INFO
    sport VARCHAR(50) NOT NULL,
    location VARCHAR(150) NOT NULL,
    match_date DATE NOT NULL,
    match_time TIME NOT NULL,

    -- PLAYERS
    current_players INT DEFAULT 1,
    max_players INT NOT NULL,

    -- DETAILS
    description TEXT,

    -- STATUS
    status VARCHAR(20) DEFAULT 'open',
    type VARCHAR(20) DEFAULT 'offline',

    -- EMAIL
    email_sent BOOLEAN DEFAULT FALSE,

    -- UPDATE NOTE
    update_note TEXT,
    update_note_updated_at TIMESTAMP,

    -- TIMESTAMPS
    created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP
);



3)participants table
CREATE TABLE match_participants (
    id SERIAL PRIMARY KEY,

    match_id INTEGER REFERENCES matches(id) ON DELETE CASCADE,

    user_id INTEGER REFERENCES users(id) ON DELETE CASCADE,

    gaming_match_id INTEGER,

    joined_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,

    UNIQUE(match_id, user_id)
);

4)gaming-groups query
CREATE TABLE gaming_groups (
    id SERIAL PRIMARY KEY,

    -- ADMIN INFO
    admin_name VARCHAR(100) NOT NULL,
    admin_email VARCHAR(100) NOT NULL,

    -- GAME INFO
    game VARCHAR(100) NOT NULL,
    platform VARCHAR(50),

    -- EVENT INFO
    event_date DATE NOT NULL,
    event_time TIME NOT NULL,

    -- PLAYERS
    current_players INT DEFAULT 1,
    max_players INT NOT NULL,

    -- JOIN INFO
    discord_link TEXT,
    room_code VARCHAR(100),

    -- DETAILS
    description TEXT,

    -- STATUS
    status VARCHAR(20) DEFAULT 'open',
    type VARCHAR(20) DEFAULT 'gaming',

    -- EMAIL
    email_sent BOOLEAN DEFAULT FALSE,

    -- UPDATE NOTE
    update_note TEXT,
    update_note_updated_at TIMESTAMP,

    -- TIMESTAMPS
    created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP
);
